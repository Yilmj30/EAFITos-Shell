113static const HelpEntry* find_help_by_any_alias(const char *token) {
114 if (!token) return NULL;
115
116 for (int i = 0; i < help_len(); i++) {
117 const HelpEntry *h = &HELP[i];
118
119 /* Comparar contra aliases ES/EN/FR en la tabla de comandos */
120 for (int j = 0; j < num_comandos(); j++) {
121 if (str_eq(h->es, tabla_comandos[j].es)) {
122 if (str_eq(token, tabla_comandos[j].es) ||
123 str_eq(token, tabla_comandos[j].en) ||
124 str_eq(token, tabla_comandos[j].fr)) {
125 return h;
126 }
127 }
128 }
129 }
130 return NULL;
131}
